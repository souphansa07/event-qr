ເຊື່ອມຕໍ່ Google Sheets (Real-time)

ບັນທຶກຂໍ້ມູນລົງທະບຽນ ແລະ ສະຖານະສະແກນເຂົ້າງານລົງ Google Sheets ອັດຕະໂນມັດ

* ລິ້ງທີ່ຖືກຕ້ອງຕ້ອງຂຶ້ນຕົ້ນດ້ວຍ https://script.google.com/macros/s/... ແລະ ລົງທ້າຍດ້ວຍ /exec

2. ວິທີຕັ້ງຄ່າໃນ Google Sheets (ເຮັດຄັ້ງດຽວ ໃຊ້ເວລາ 1 ນາທີ):
  1. ສ້າງໄຟລ໌ Google Sheets ໃໝ່ ແລ້ວໄປທີ່ເມນູ Extensions (ສ່ວນຂະຫຍາຍ) > Apps Script
  2. ລຶບໂຄ້ດເກົ່າອອກໃຫ້ໝົດ ແລ້ວວາງໂຄ້ດດ້ານລຸ່ມນີ້ລົງໄປ ແລ້ວກົດປຸ່ມ Save (ບັນທຶກ)
  3. ກົດປຸ່ມ Deploy (ເຮັດໃຫ້ໃຊ້ງານໄດ້) > New deployment
  4. ເລືອກ Type ເປັນ Web app, ຕັ້ງ Execute as: Me ແລະ Who has access: Anyone ແລ້ວກົດ Deploy
  5. ຄັດລອກ Web app URL ມາວາງໃສ່ຊ່ອງດ້ານເທິງນີ້ແລ້ວກົດບັນທຶກ!
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ລະຫັດລົງທະບຽນ", "ສະຖານະເຂົ້າງານ", "ເວລາເຂົ້າງານ", "ຊື່", "ນາມສະກຸນ", "ສາຂາ", "ປີຮຽນ", "ເບີໂທ", "ວັນທີລົງທະບຽນ"]);
  }
  var data = JSON.parse(e.postData.contents);
  var rows = sheet.getDataRange().getValues();
  
  // ກວດສອບວ່າມີລະຫັດນີ້ແລ້ວຫຼືຍັງ (ຖ້າມີໃຫ້ອັບເດດສະຖານະເຂົ້າງານ)
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][0] === data.regId) {
      sheet.getRange(i + 1, 2).setValue(data.checkedIn ? "ເຂົ້າງານແລ້ວ" : "ຍັງບໍ່ເຂົ້າງານ");
      sheet.getRange(i + 1, 3).setValue(data.checkInTime || "-");
      return ContentService.createTextOutput(JSON.stringify({status: "updated"})).setMimeType(ContentService.MimeType.JSON);
    }
  }
  
  // ເພີ່ມແຖວໃໝ່
  sheet.appendRow([
    data.regId,
    data.checkedIn ? "ເຂົ້າງານແລ້ວ" : "ຍັງບໍ່ເຂົ້າງານ",
    data.checkInTime || "-",
    data.firstName,
    data.lastName,
    data.major,
    data.studyYear,
    "'" + data.phone,
    data.timestamp
  ]);
  return ContentService.createTextOutput(JSON.stringify({status: "added"})).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var result = [];
  for (var i = 1; i < rows.length; i++) {
    if (!rows[i][0]) continue;
    result.push({
      uid: String(rows[i][0]),
      regId: String(rows[i][0]),
      checkedIn: rows[i][1] === "ເຂົ້າງານແລ້ວ",
      checkInTime: rows[i][2] === "-" ? null : String(rows[i][2]),
      firstName: String(rows[i][3] || ""),
      lastName: String(rows[i][4] || ""),
      fullName: (String(rows[i][3] || "") + " " + String(rows[i][4] || "")).trim(),
      major: String(rows[i][5] || ""),
      studyYear: String(rows[i][6] || ""),
      phone: String(rows[i][7] || "").replace(/^'/, ""),
      timestamp: String(rows[i][8] || ""),
      eventTitle: "ແບບຟອມລົງທະບຽນເຂົ້າຮ່ວມງານ"
    });
  }
  return ContentService.createTextOutput(JSON.stringify(result.reverse())).setMimeType(ContentService.MimeType.JSON);
}
